# The kernel decides a module's flags, so this does not compile anything itself
# - it hands the whole directory to kbuild and tracks the result. What it buys
# is that `cmake --build build` produces the module alongside the binaries
# rather than leaving it to a second, separate command that is easy to forget.

set(RAILFS_KDIR "/lib/modules/${CMAKE_SYSTEM_VERSION}/build"
    CACHE PATH "Kernel build tree the module is compiled against")

# A machine without the matching kernel headers builds everything else and says
# so, rather than failing the whole tree for a module it cannot compile.
if(NOT EXISTS ${RAILFS_KDIR}/Makefile)
  message(WARNING "no kernel build tree at ${RAILFS_KDIR}; railfs.ko will not build")
  return()
endif()

find_program(RAILFS_MAKE make REQUIRED)
find_program(RAILFS_OPENSSL openssl)

set(RAILFS_OBJECTS railfs-main.o railfs-read.o railfs-write.o railfs-direct.o railfs-lookup.o railfs-create.o railfs-mkdir.o railfs-unlink.o railfs-rename.o railfs-symlink.o railfs-link.o railfs-readdir.o railfs-open.o railfs-fsync.o railfs-setattr.o railfs-statfs.o railfs-inode.o railfs-proto.o railfs-tcp.o railfs-rdma.o railfs-gds.o railfs-trace.o)

set(RAILFS_SOURCES
    railfs-main.c
    railfs-read.c
    railfs-write.c
    railfs-direct.c
    railfs-lookup.c
    railfs-create.c
    railfs-mkdir.c
    railfs-unlink.c
    railfs-rename.c
    railfs-symlink.c
    railfs-link.c
    railfs-readdir.c
    railfs-open.c
    railfs-fsync.c
    railfs-setattr.c
    railfs-statfs.c
    railfs-inode.c
    railfs.h
    railfs-proto.c
    railfs-tcp.c
    railfs-rdma.c
    railfs-gds.c
    railfs-trace.c
    railfs-msg.h
    railfs-proto.h
    railfs-tcp.h
    railfs-rdma.h
    railfs-gds.h
    railfs-nvfs.h
    railfs-trace.h)

# kbuild reads this from src=, so it has to sit beside the sources. Generated
# rather than checked in, so the object list lives in one place.
string(REPLACE ";" " " RAILFS_OBJECT_LINE "${RAILFS_OBJECTS}")
file(WRITE ${CMAKE_CURRENT_SOURCE_DIR}/Kbuild
     "# Generated by cmake from src/linux/CMakeLists.txt. Do not edit.\n"
     "obj-m := railfs.o\n"
     "railfs-y := ${RAILFS_OBJECT_LINE}\n")

# Where the module lands. Out of the source tree when the kernel can do it,
# because deploy rsyncs with --delete and a module built beside its sources is
# removed on every deploy. KBUILD_EXTMOD_OUTPUT is the supported way; passing
# src= instead, as some examples do, is not honoured here and fails with
# "No rule to make target 'railfs-main.o'".
file(READ ${RAILFS_KDIR}/Makefile RAILFS_KERNEL_MAKEFILE LIMIT 200000)
string(FIND "${RAILFS_KERNEL_MAKEFILE}" "KBUILD_EXTMOD_OUTPUT" RAILFS_HAS_EXTMOD_OUTPUT)

if(RAILFS_HAS_EXTMOD_OUTPUT GREATER -1)
  set(RAILFS_MODULE ${CMAKE_CURRENT_BINARY_DIR}/railfs.ko)
  set(RAILFS_OUTPUT_ARG KBUILD_EXTMOD_OUTPUT=${CMAKE_CURRENT_BINARY_DIR})
else()
  message(STATUS "kernel predates KBUILD_EXTMOD_OUTPUT; railfs.ko is built beside its sources")
  set(RAILFS_MODULE ${CMAKE_CURRENT_SOURCE_DIR}/railfs.ko)
  set(RAILFS_OUTPUT_ARG "")
endif()

add_custom_command(
  OUTPUT ${RAILFS_MODULE}
  COMMAND ${RAILFS_MAKE} -C ${RAILFS_KDIR}
          M=${CMAKE_CURRENT_SOURCE_DIR}
          ${RAILFS_OUTPUT_ARG}
          modules
  DEPENDS ${RAILFS_SOURCES}
  COMMENT "Building railfs.ko against ${RAILFS_KDIR}"
  VERBATIM)

add_custom_target(railfs-module ALL DEPENDS ${RAILFS_MODULE})

# Secure Boot refuses an unsigned module. These were the Makefile's `sign` and
# `mok` targets; they are not part of a build, so neither is in ALL.
set(RAILFS_MOK_KEY ${CMAKE_CURRENT_SOURCE_DIR}/MOK.priv)
set(RAILFS_MOK_CERT ${CMAKE_CURRENT_SOURCE_DIR}/MOK.der)

add_custom_target(railfs-sign
  COMMAND test -f ${RAILFS_MOK_KEY} ||
          { echo "no signing key at ${RAILFS_MOK_KEY}; build the railfs-mok target first" && exit 1 ; }
  COMMAND ${RAILFS_KDIR}/scripts/sign-file sha256 ${RAILFS_MOK_KEY} ${RAILFS_MOK_CERT} ${RAILFS_MODULE}
  COMMAND modinfo ${RAILFS_MODULE} | grep -E "^sig" || echo "warning: no signature recorded"
  DEPENDS railfs-module
  COMMENT "Signing railfs.ko with the enrolled key"
  VERBATIM)

# One key per machine, kept out of the repo. Enrolling it is a separate,
# deliberate step: mokutil --import, then the console at the next boot.
add_custom_target(railfs-mok
  COMMAND test ! -f ${RAILFS_MOK_KEY} ||
          { echo "${RAILFS_MOK_KEY} already exists, refusing to replace it" && exit 1 ; }
  COMMAND ${RAILFS_OPENSSL} req -new -x509 -newkey rsa:2048 -nodes -days 36500
          -keyout ${RAILFS_MOK_KEY} -outform DER -out ${RAILFS_MOK_CERT}
          -subj "/CN=railfs module signing/"
  COMMAND chmod 600 ${RAILFS_MOK_KEY}
  COMMENT "Creating a module signing key for this machine"
  VERBATIM)
